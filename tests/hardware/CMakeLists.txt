cmake_minimum_required(VERSION 3.20.0)
if(NOT DEFINED EXTRA_CONF_FILE)
  set(EXTRA_CONF_FILE profiles/core.conf CACHE STRING "Hardware acceptance configuration")
endif()
if(NOT DEFINED DTC_OVERLAY_FILE)
  set(DTC_OVERLAY_FILE profiles/core.overlay CACHE STRING "Hardware acceptance overlay")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(hpm5361_hardware_acceptance)
target_sources(app PRIVATE src/main.c)
foreach(profile core counter_capture reset_hwinfo dma flash watchdog can ptpc motor gpio uart lin i2c spi
		analog front_end pwm qei)
  string(TOUPPER ${profile} symbol)
  target_sources_ifdef(CONFIG_HPM_TEST_${symbol} app PRIVATE src/${profile}.c)
endforeach()

target_sources_ifdef(CONFIG_HPM_TEST_UART_ADDRESS app PRIVATE src/uart_address.c)
target_sources_ifdef(CONFIG_HPM_TEST_UART_RS485 app PRIVATE src/uart_rs485.c)
target_sources_ifdef(CONFIG_HPM_TEST_I2C_FAST app PRIVATE src/i2c.c)
target_sources_ifdef(CONFIG_HPM_TEST_I2C_FAST_PLUS app PRIVATE src/i2c.c)
target_sources_ifdef(CONFIG_HPM_TEST_QEI_PD app PRIVATE src/qei_pd.c)
target_sources_ifdef(CONFIG_HPM_TEST_QEI_UD app PRIVATE src/qei_ud.c)
target_sources_ifdef(CONFIG_HPM_TEST_QEI_SINGLE app PRIVATE src/qei_single.c)
target_sources_ifdef(CONFIG_HPM_TEST_QEI_UVW app PRIVATE src/qei_uvw.c)
target_sources_ifdef(CONFIG_HPM_TEST_USB_DEVICE app PRIVATE src/usb_device.c)
target_sources_ifdef(CONFIG_HPM_TEST_USB_ENDPOINTS app PRIVATE src/usb_endpoints.c)
target_sources_ifdef(CONFIG_HPM_TEST_USB_HOST app PRIVATE src/usb_host.c)
target_sources_ifdef(CONFIG_HPM_TEST_USB_HOST_HID_REPORT app PRIVATE src/usb_host_hid_report.c)

if(CONFIG_HPM_TEST_USB_HOST OR CONFIG_HPM_TEST_USB_HOST_HID_REPORT)
  target_include_directories(app PRIVATE ${ZEPHYR_BASE}/subsys/usb/host)
endif()

if(CONFIG_HPM_TEST_CRYPTO)
  target_sources(app PRIVATE ../hpm5300evk/src/crypto.c)
endif()
